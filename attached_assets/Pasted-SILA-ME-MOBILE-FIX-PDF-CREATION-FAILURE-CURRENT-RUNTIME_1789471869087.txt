2. CREATE A URI-AWARE IMAGE READER
==================================================

Refactor readImageBytes() so it supports the actual URI type.

Conceptually:

async function readImageBytes(uri: string): Promise<Uint8Array> {

    if (!uri) {
        throw new Error("EMPTY_IMAGE_URI");
    }

    // WEB / HTTP / BLOB
    if (
        uri.startsWith("blob:") ||
        uri.startsWith("http://") ||
        uri.startsWith("https://")
    ) {
        const response = await fetch(uri);

        if (!response.ok && !uri.startsWith("blob:")) {
            throw new Error(
              `IMAGE_FETCH_FAILED:${response.status}`
            );
        }

        const buffer = await response.arrayBuffer();

        return new Uint8Array(buffer);
    }

    // DATA URI
    if (uri.startsWith("data:")) {
        decode the base64 payload safely
        and return Uint8Array.
    }

    // NATIVE LOCAL FILE
    if (uri.startsWith("file://")) {
        read through the supported Expo FileSystem API.
        return Uint8Array.
    }

    // ANDROID CONTENT URI
    if (uri.startsWith("content://")) {
        first copy/materialize it into app cache as a file:// URI
        using the supported Expo/native file API,
        then read the cached file bytes.
    }

    // PLAIN ABSOLUTE NATIVE PATH
    if (uri.startsWith("/")) {
        normalize to file:// if required
        then read.
    }

    throw new Error(
      `UNSUPPORTED_IMAGE_URI_SCHEME:${uri.substring(0, 30)}`
    );
}

Use the APIs appropriate to the CURRENT installed Expo SDK.

Do not blindly paste deprecated Expo FileSystem code.
